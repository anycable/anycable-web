# FastAPI: stream an LLM answer the client can resume
import json, httpx

BROADCAST_URL = "http://anycable:8090/_broadcast"
AUTH = {"Authorization": f"Bearer {BROADCAST_KEY}"}

async def stream_answer(chat_id: str, prompt: str):
    # Runs apart from the request, so a closed tab can't stop it
    async with httpx.AsyncClient(headers=AUTH) as cable:

            await cable.post(BROADCAST_URL, json={
                "stream": f"chat/{chat_id}",
