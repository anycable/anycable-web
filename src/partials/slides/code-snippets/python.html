# FastAPI: stream an LLM answer the client can resume
import base64, hashlib, hmac, json, os, httpx

SECRET = os.environ["ANYCABLE_SECRET"].encode()
BROADCAST_URL = "http://anycable:8080/_broadcast"

def hmac_hex(message: bytes) -> str:
    return hmac.new(SECRET, message, hashlib.sha256).hexdigest()

BROADCAST_KEY = hmac_hex(b"broadcast-cable")

def sign_stream(name: str) -> str:
    encoded = base64.b64encode(json.dumps(name).encode()).decode()
    return f"{encoded}--{hmac_hex(encoded.encode())}"

async def stream_answer(chat_id: str, prompt: str):
    # Runs apart from the request, so a closed tab can't stop it
    auth = {"Authorization": f"Bearer {BROADCAST_KEY}"}
    async with httpx.AsyncClient(headers=auth) as cable:
        async for token in llm.stream(prompt):
            await cable.post(BROADCAST_URL, json={
                "stream": f"chat/{chat_id}",
                "data": json.dumps({"token": token}),
            })

@app.post("/chats/{chat_id}/answer")
async def answer(chat_id: str, q: Ask, bg: BackgroundTasks):
    bg.add_task(stream_answer, chat_id, q.prompt)
    # The browser subscribes with streamFromSigned(); AnyCable
    # replays missed tokens when it reconnects
    return {"stream": sign_stream(f"chat/{chat_id}")}
